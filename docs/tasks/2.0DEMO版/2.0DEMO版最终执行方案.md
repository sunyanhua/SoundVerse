基于原始SoundVerse系统，创建一个核心功能精简、部署便捷的2.0 DEMO版本，专注于：
- 默认登录（免认证）
- 音频上传与智能裁切
- 语弹库管理
- AI对话实验室（带点赞保存功能）
- 一键部署与阿里云SLB适配

```python
class PresetPrompt(Base):
    """
    预置提示词表（点赞保存）
    """
    __tablename__ = "preset_prompts"
    
    # 基础信息
    id = Column(String(36), primary_key=True, default=lambda: str(uuid.uuid4()))
    user_id = Column(String(36), ForeignKey("users.id"), index=True, nullable=False)
    original_message_id = Column(String(36), ForeignKey("chat_messages.id"), nullable=True)
    
    # 提示词内容
    query_text = Column(Text, nullable=False)  # 用户原始查询
    category = Column(String(50), nullable=True)  # 分类（对话、音乐、学习等）
    emotion = Column(String(50), nullable=True)  # 情感标签
    tags = Column(JSON, nullable=True)  # 标签数组
    
    # 使用统计
    use_count = Column(Integer, default=0)  # 被使用次数
    like_count = Column(Integer, default=1)  # 点赞次数（默认1）
    
    # 审核状态
    review_status = Column(
        String(20),
        default="pending",  # pending, approved, rejected
        nullable=False
    )
    
    # 时间戳
    created_at = Column(DateTime, default=datetime.utcnow)
    updated_at = Column(DateTime, default=datetime.utcnow, onupdate=datetime.utcnow)
    
    # 关系
    user = relationship("User")
    original_message = relationship("ChatMessage")
    
    def __repr__(self) -> str:
        return f"<PresetPrompt(id={self.id}, query='{self.query_text[:50]}...')>"
```

```python
@router.post("/preset-prompts")
async def create_preset_prompt(
    request: dict,
    current_user: User = Depends(get_current_user_or_mock),
    db: AsyncSession = Depends(get_db),
) -> dict:
    """
    创建预置提示词（点赞保存）
    request: {
        "query_text": "今天天气怎么样？",
        "original_message_id": "optional",
        "category": "weather",
        "emotion": "curious",
        "tags": ["天气", "日常"]
    }
    """
    try:
        from shared.models.chat import PresetPrompt
        
        preset = PresetPrompt(
            user_id=current_user.id,
            query_text=request.get("query_text"),
            original_message_id=request.get("original_message_id"),
            category=request.get("category"),
            emotion=request.get("emotion"),
            tags=request.get("tags", []),
            review_status="approved",  # DEMO版本直接通过
        )
        
        db.add(preset)
        await db.commit()
        await db.refresh(preset)
        
        return {
            "success": True,
            "id": preset.id,
            "message": "提示词已保存"
        }
    except Exception as e:
        logger.error(f"创建预置提示词失败: {str(e)}")
        raise HTTPException(
            status_code=status.HTTP_500_INTERNAL_SERVER_ERROR,
            detail="保存提示词失败",
        )


@router.get("/preset-prompts/random")
async def get_random_preset_prompts(
    count: int = 5,
    category: Optional[str] = None,
    current_user: User = Depends(get_current_user_or_mock),
    db: AsyncSession = Depends(get_db),
) -> dict:
    """
    随机获取预置提示词（用于"换一批"功能）
    """
    try:
        from sqlalchemy import select, func
        from shared.models.chat import PresetPrompt
        
        query = select(PresetPrompt).where(
            PresetPrompt.review_status == "approved"
        )
        
        if category:
            query = query.where(PresetPrompt.category == category)
        
        # 随机排序
        query = query.order_by(func.rand()).limit(count)
        
        result = await db.execute(query)
        prompts = result.scalars().all()
        
        return {
            "prompts": [
                {
                    "id": p.id,
                    "query_text": p.query_text,
                    "category": p.category,
                    "emotion": p.emotion,
                    "tags": p.tags,
                    "use_count": p.use_count,
                    "like_count": p.like_count,
                }
                for p in prompts
            ]
        }
    except Exception as e:
        logger.error(f"获取随机提示词失败: {str(e)}")
        raise HTTPException(
            status_code=status.HTTP_500_INTERNAL_SERVER_ERROR,
            detail="获取提示词失败",
        )


@router.put("/messages/{message_id}/like")
async def like_message(
    message_id: str,
    save_as_prompt: bool = True,  # 是否保存为预置提示词
    current_user: User = Depends(get_current_user_or_mock),
    db: AsyncSession = Depends(get_db),
) -> dict:
    """
    点赞消息并可选保存为预置提示词
    """
    try:
        from sqlalchemy import select
        from shared.models.chat import ChatMessage
        
        # 1. 更新消息反馈
        result = await db.execute(
            select(ChatMessage).where(
                ChatMessage.id == message_id,
                ChatMessage.role == "user"  # 只允许点赞用户消息
            )
        )
        message = result.scalar_one_or_none()
        
        if not message:
            raise HTTPException(
                status_code=status.HTTP_404_NOT_FOUND,
                detail="消息不存在",
            )
        
        message.set_feedback("like", "用户点赞")
        await db.commit()
        
        # 2. 如果保存为预置提示词
        if save_as_prompt:
            from shared.models.chat import PresetPrompt
            
            preset = PresetPrompt(
                user_id=current_user.id,
                original_message_id=message_id,
                query_text=message.content,
                category="conversation",  # 默认分类
                emotion=message.emotion,
                tags=["点赞保存"],  # 默认标签
                review_status="approved",
            )
            
            db.add(preset)
            await db.commit()
        
        return {
            "success": True,
            "message": "点赞成功",
            "saved_as_prompt": save_as_prompt,
        }
    except HTTPException:
        raise
    except Exception as e:
        logger.error(f"点赞消息失败: {str(e)}")
        raise HTTPException(
            status_code=status.HTTP_500_INTERNAL_SERVER_ERROR,
            detail="点赞失败",
        )
```

### 3.1 部署脚本结构

创建`deploy.sh`：

```bash
#!/bin/bash

# SoundVerse 2.0 DEMO 一键部署脚本
# 使用方法: ./deploy.sh [环境: dev/prod] [服务器IP]

set -e  # 遇到错误立即退出

# 配置参数
ENV=${1:-"prod"}
SERVER_IP=${2:-"your-server-ip"}
APP_NAME="soundverse-demo"
BACKUP_DIR="./backups"
TIMESTAMP=$(date +"%Y%m%d_%H%M%S")

# 颜色输出
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
NC='\033[0m' # No Color

log_info() {
    echo -e "${GREEN}[INFO]${NC} $1"
}

log_warn() {
    echo -e "${YELLOW}[WARN]${NC} $1"
}

log_error() {
    echo -e "${RED}[ERROR]${NC} $1"
}

# 检查参数
if [ -z "$SERVER_IP" ] || [ "$SERVER_IP" = "your-server-ip" ]; then
    log_error "请指定服务器IP地址: ./deploy.sh prod 1.2.3.4"
    exit 1
fi

# 步骤1: 本地代码打包
log_info "步骤1: 本地代码检查与打包..."

# 检查必要文件
required_files=(
    "docker-compose.yml"
    "frontend-demo/Dockerfile"
    "backend/Dockerfile.dev"
    ".env.example"
)

for file in "${required_files[@]}"; do
    if [ ! -f "$file" ]; then
        log_error "缺少必要文件: $file"
        exit 1
    fi
done

# 复制环境文件
if [ ! -f ".env" ]; then
    if [ -f ".env.example" ]; then
        cp .env.example .env
        log_warn "已创建.env文件，请根据实际情况修改配置"
    else
        log_error "缺少.env.example文件"
        exit 1
    fi
fi

# 创建备份目录
mkdir -p "$BACKUP_DIR"

# 步骤2: 数据库备份（如果已存在）
log_info "步骤2: 检查远程数据库备份..."
if ssh root@$SERVER_IP "docker ps | grep -q mysql" 2>/dev/null; then
    log_info "检测到现有MySQL容器，执行备份..."
    
    # 创建备份
    BACKUP_FILE="${APP_NAME}_db_backup_${TIMESTAMP}.sql"
    ssh root@$SERVER_IP "docker exec ${APP_NAME}-mysql mysqldump -u soundverse -ppassword soundverse > /tmp/${BACKUP_FILE} 2>/dev/null || true"
    
    # 下载备份
    scp root@$SERVER_IP:/tmp/${BACKUP_FILE} ${BACKUP_DIR}/${BACKUP_FILE}
    
    if [ -s "${BACKUP_DIR}/${BACKUP_FILE}" ]; then
        log_info "数据库备份完成: ${BACKUP_DIR}/${BACKUP_FILE}"
    else
        log_warn "数据库备份可能为空或失败"
    fi
fi

# 步骤3: SSH同步代码到服务器
log_info "步骤3: 同步代码到服务器 $SERVER_IP ..."

# 创建服务器目录结构
ssh root@$SERVER_IP "mkdir -p /opt/${APP_NAME}/{backend,frontend-demo,data}"

# 同步关键文件
rsync -avz --progress \
    --exclude='.git' \
    --exclude='node_modules' \
    --exclude='__pycache__' \
    --exclude='*.pyc' \
    --exclude='.env' \
    --exclude='.DS_Store' \
    ./ root@$SERVER_IP:/opt/${APP_NAME}/

# 上传环境文件（如果本地存在）
if [ -f ".env" ]; then
    scp .env root@$SERVER_IP:/opt/${APP_NAME}/.env
    log_info "已上传环境配置文件"
else
    log_warn "未找到.env文件，将使用服务器现有配置"
fi

# 步骤4: 远程构建与启动
log_info "步骤4: 在服务器上构建并启动服务..."

# 创建启动脚本
cat > /tmp/start_soundverse.sh << 'EOF'

set -e

APP_NAME="soundverse-demo"
WORK_DIR="/opt/${APP_NAME}"

cd "$WORK_DIR"

# 检查Docker Compose是否可用
if ! command -v docker-compose &> /dev/null; then
    # 尝试使用docker compose插件
    if docker compose version &> /dev/null; then
        COMPOSE_CMD="docker compose"
    else
        echo "错误: 未找到docker-compose或docker compose插件"
        exit 1
    fi
else
    COMPOSE_CMD="docker-compose"
fi

# 停止现有服务
echo "停止现有服务..."
$COMPOSE_CMD down --remove-orphans || true

echo "构建新服务..."

echo "启动服务..."

# 检查服务状态

# 上传并执行启动脚本
scp /tmp/start_soundverse.sh root@$SERVER_IP:/opt/${APP_NAME}/

# 步骤5: 健康检查
log_info "步骤5: 执行健康检查..."

# 检查前端服务

    log_info "前端服务健康: http://${SERVER_IP}:5173"

# 检查后端API

fi

# 步骤6: 清理和报告
log_info "步骤6: 部署完成！"

echo ""
echo "================================"
echo "部署成功！"
echo "环境: $ENV"
```

#### 4.4.3 阿里云CLI配置脚本
创建`slb-config.sh`：

```bash
#!/bin/bash

# 阿里云SLB自动配置脚本
# 需要安装aliyun-cli并配置AK/SK

SLB_ID="lb-xxx"
SERVER_ID="i-xxx"
SERVER_IP="1.2.3.4"

# 1. 创建HTTP监听（前端）
aliyun slb CreateLoadBalancerTCPListener \
  --LoadBalancerId $SLB_ID \
  --ListenerPort 80 \
  --BackendServerPort 5173 \
  --Bandwidth -1 \
  --Scheduler wrr \
  --PersistenceTimeout 0 \
  --HealthCheckType tcp \
  --HealthCheckDomain "$SERVER_IP" \
  --HealthCheckURI "/" \
  --HealthyThreshold 3 \
  --UnhealthyThreshold 3 \
  --HealthCheckTimeout 5 \
  --HealthCheckInterval 2

# 2. 创建API监听（后端）
aliyun slb CreateLoadBalancerTCPListener \
  --LoadBalancerId $SLB_ID \
  --ListenerPort 8000 \
  --BackendServerPort 8000 \
  --Bandwidth -1 \
  --Scheduler wrr \
  --PersistenceTimeout 0 \
  --HealthCheckType tcp \
  --HealthCheckDomain "$SERVER_IP" \
  --HealthCheckURI "/api/health" \
  --HealthyThreshold 3 \
  --UnhealthyThreshold 3 \
  --HealthCheckTimeout 5 \
  --HealthCheckInterval 2

# 3. 添加后端服务器
aliyun slb AddBackendServers \
  --LoadBalancerId $SLB_ID \
  --BackendServers "[{\"ServerId\":\"$SERVER_ID\",\"Weight\":\"100\"}]"

echo "SLB配置完成！"
```

   ```
   检查: curl http://服务器IP:8000/api/health

   ```

   ```
   检查: 后端日志，数据库preset_prompts表
   解决: 确认数据库迁移已执行
   ```
